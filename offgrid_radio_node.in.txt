#include <Arduino.h>
#include <SPI.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <RadioLib.h> 
#include <driver/i2s.h> 

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// Hardware & Peripheral Layout Pins
const int BTNS[] = {2, 3, 4}; // LEFT, MID, RIGHT
const int POT_PIN = 1, BUZZER_PIN = 9;
const int I2S_MIC[] = {42, 41, 40}, I2S_SPK[] = {39, 38, 37};
const int LORA_NSS = 7, LORA_RXEN = 13, LORA_TXEN = 14, LORA_DIO1 = 5, LORA_NRST = 8, LORA_BUSY = 6;

SX1262 radio = new Module(LORA_NSS, LORA_DIO1, LORA_NRST, LORA_BUSY);

#define AUDIO_SAMPLE_RATE 8000 
#define AUDIO_BUFFER_SIZE (AUDIO_SAMPLE_RATE * 10) 
uint8_t* voiceStorageBuffer = NULL; 
uint32_t activeAudioDataSize = 0;

enum SystemState {
  STATE_WELCOME, STATE_MAIN_HUB, STATE_SERVER_SELECT, STATE_ESP_SELECT, STATE_ACTION_SELECT,
  STATE_VOICE_RECORD, STATE_TEXT_MENU, STATE_TEXT_CUSTOM, STATE_INBOX_MENU, STATE_INBOX_TEXT,
  STATE_INBOX_VOICE, STATE_VOICE_REQUEST, STATE_GPS_ENV_PROMPT, STATE_GPS_MENU, STATE_GPS_ROSTER,
  STATE_GPS_DETAILS, STATE_HW_CONFIG, STATE_MAIL_POPUP       
};
SystemState currentState = STATE_WELCOME, prePopupState = STATE_MAIN_HUB; 

// Combined Menu Structures to reduce array sprawl
const char* mainHubOptions[] = {"[ SEND MESSAGES ]", "[ OPEN INBOX ]", "[ RSSI RADIUS TRACK ]", "[ POWER & HW CONFIG ]"};
const char* serverList[]     = {"Server ALPHA", "Server BRAVO", "Server OMEGA"};
const char* espList[]        = {"[ BROADCAST ALL ]", "ESP32-S3 #996", "ESP32-S3 #126", "ESP32-S3 #441"};
const char* actionList[]     = {"Send Voice Note", "Send Text Msg"};
const char* textMenuList[]   = {"I NEED HELP", "I AM SAFE", "RESCUE ME", "[ TYPE CUSTOM MSG ]"};
const char* inboxChoices[]   = {"View Text Messages", "View Voice Notes"};
const char* storedTexts[]    = {"#996: Moving to ridge", "#126: Battery at 15%", "ALPHA: Stay put"};

int currentHubIdx = 0, currentServerIdx = 0, currentEspIdx = 0, currentActionIdx = 0;
int currentTextOptIdx = 0, currentInboxChoiceIdx = 0, currentTextMsgIdx = 0, totalStoredTexts = 3;

bool localVoiceNoteExists = false, isPlayingVoiceNote = false, isIndoorEnvironment = false, trackingEngineActive = true;
int currentTxPowerDbm = 10, currentLetterIdx = 0;
float calculatedMaxRangeKm = 0.0;
unsigned long lastActivityTime = 0, popupStartTime = 0, buttonPressStartTime = 0, midBtnPressTime = 0;
int activePressedPin = -1;
bool midBtnActive = false;
String customMessage = "";
const char alphabet[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789 <-";

struct PeerDevice { const char* id; float radius, liveRssi, liveSnr; int heading; const char* compass; unsigned long lastSeenMs; };
PeerDevice trackerRoster[] = { {"ESP32-S3 #996", 0.0, -120.0, 0.0, 45, "NE", 0}, {"ESP32-S3 #126", 0.0, -120.0, 0.0, 270, "W", 0}, {"ESP32-S3 #441", 0.0, -120.0, 0.0, 195, "SSW", 0} };

void handleWelcomeAnimation();
void checkButtons();
void updateUI();
void drawHeader(const char* title);
void drawSelector(int currentIdx, int totalItems, const char* items[]);
void drawVoiceUI(unsigned long elapsed);
void handleTransmitText(const char* payload);
void processIncomingRadioTraffic();
float calculateRadiusFromRssi(float rssi, bool indoor);
void readHardwarePotentiometer();
void wakeDisplay();
void checkPowerTimeout();
void triggerAlertBeep();
void initHardwareAudioI2S();
void recordVoiceNoteBuffer();
void playActiveVoiceBuffer();
void transmitVoicePacketOverAir();
void parseNetworkPacketPayload(String textData, float rssi, float snr);

void setup() {
  Serial.begin(115200);
  for (int p : BTNS) pinMode(p, INPUT_PULLUP);
  pinMode(POT_PIN, INPUT); pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LORA_RXEN, OUTPUT); pinMode(LORA_TXEN, OUTPUT);

  if (!(voiceStorageBuffer = (uint8_t*)malloc(AUDIO_BUFFER_SIZE)) || !display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) while(1);

  display.clearDisplay(); display.setTextColor(SSD1306_WHITE);
  handleWelcomeAnimation(); initHardwareAudioI2S();

  if (radio.begin(868.0, 125.0, 9, 7, 0x12, currentTxPowerDbm, 8) == RADIOLIB_ERR_NONE) {
    handleTransmitText("[SYS_INIT]: NODE_ONLINE");
    radio.startReceive();
    digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
  } else {
    while(1);
  }
  lastActivityTime = millis(); currentState = STATE_MAIN_HUB;
}

void loop() {
  checkButtons();
  if (trackingEngineActive) processIncomingRadioTraffic();
  readHardwarePotentiometer(); checkPowerTimeout(); updateUI();
  delay(10);
}

void initHardwareAudioI2S() {
  i2s_config_t micConf = { (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX), AUDIO_SAMPLE_RATE, I2S_BITS_PER_SAMPLE_16BIT, I2S_CHANNEL_FMT_ONLY_LEFT, I2S_COMM_FORMAT_STAND_I2S, ESP_INTR_FLAG_LEVEL1, 4, 512, false };
  i2s_pin_config_t micPins = { I2S_MIC[2], I2S_MIC[0], -1, I2S_MIC[1] };
  i2s_driver_install(I2S_NUM_0, &micConf, 0, NULL); i2s_set_pin(I2S_NUM_0, &micPins);
}

void recordVoiceNoteBuffer() {
  size_t read = 0; unsigned long start = millis(); activeAudioDataSize = 0;
  while(millis() - start < 10000 && activeAudioDataSize < AUDIO_BUFFER_SIZE - 1024) {
    i2s_read(I2S_NUM_0, (void*)(voiceStorageBuffer + activeAudioDataSize), 1024, &read, portMAX_DELAY);
    activeAudioDataSize += read; drawVoiceUI(millis() - start); display.display();
  }
  localVoiceNoteExists = true;
}

void playActiveVoiceBuffer() {
  if (!localVoiceNoteExists || !activeAudioDataSize) return;
  size_t written = 0; uint32_t ptr = 0; i2s_zero_dma_buffer(I2S_NUM_0);
  while(ptr < activeAudioDataSize && isPlayingVoiceNote) {
    i2s_write(I2S_NUM_0, (void*)(voiceStorageBuffer + ptr), 1024, &written, portMAX_DELAY); ptr += written;
    if(digitalRead(BTNS[1]) == LOW) break;
  }
  isPlayingVoiceNote = false; i2s_zero_dma_buffer(I2S_NUM_0);
}

void transmitVoicePacketOverAir() {
  if(!activeAudioDataSize) return;
  char header[64]; snprintf(header, sizeof(header), "[V_NOTE:#%s:%u]", espList[currentEspIdx], activeAudioDataSize);
  digitalWrite(LORA_RXEN, LOW); digitalWrite(LORA_TXEN, HIGH);
  radio.transmit(header);
  radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
}

void parseNetworkPacketPayload(String textData, float rssi, float snr) {
  if (textData.indexOf("[SYS_INIT]") >= 0) return;
  for(int i = 0; i < 3; i++) {
    if(textData.indexOf(trackerRoster[i].id) >= 0 || (i == 0 && textData.indexOf("996") >= 0)) {
      trackerRoster[i].liveRssi = rssi; trackerRoster[i].liveSnr = snr; trackerRoster[i].lastSeenMs = millis();
      trackerRoster[i].radius = calculateRadiusFromRssi(rssi, isIndoorEnvironment);
      wakeDisplay(); triggerAlertBeep();
    }
  }
}

void processIncomingRadioTraffic() {
  if (digitalRead(LORA_DIO1) == HIGH) {
    String pStr = ""; int state = radio.readData(pStr);
    if (state == RADIOLIB_ERR_NONE) {
      parseNetworkPacketPayload(pStr, radio.getRSSI(), radio.getSNR());
      if (currentState != STATE_MAIL_POPUP && pStr.indexOf("[SYS_INIT]") < 0) { currentState = STATE_MAIL_POPUP; popupStartTime = millis(); }
    }
    radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
  }
}

void handleTransmitText(const char* payload) {
  char buf[128]; snprintf(buf, sizeof(buf), "[SRC:MY_NODE][DST:%s] %s", espList[currentEspIdx], payload);
  digitalWrite(LORA_RXEN, LOW); digitalWrite(LORA_TXEN, HIGH);
  radio.transmit(buf);
  radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
  lastActivityTime = millis(); delay(500);
}

void wakeDisplay() { if (!(display.readAnPins() & 0x40)) display.ssd1306_command(SSD1306_DISPLAYON); lastActivityTime = millis(); }
void checkPowerTimeout() { if (currentState != STATE_MAIL_POPUP && millis() - lastActivityTime >= 360000) display.ssd1306_command(SSD1306_DISPLAYOFF); }
void triggerAlertBeep() { for(int i=0; i<2; i++) { digitalWrite(BUZZER_PIN, HIGH); delay(150); digitalWrite(BUZZER_PIN, LOW); delay(100); } }

void readHardwarePotentiometer() {
  int p = map(analogRead(POT_PIN), 0, 4095, 2, 22);
  if (p != currentTxPowerDbm) {
    radio.setOutputPower(currentTxPowerDbm = p);
    calculatedMaxRangeKm = map(currentTxPowerDbm, 2, 22, isIndoorEnvironment ? 1 : 5, isIndoorEnvironment ? 5 : 60) / 10.0;
    wakeDisplay();
  }
}

float calculateRadiusFromRssi(float rssi, bool indoor) { return rssi >= 0 ? 0.0 : pow(10.0, (-42.0 - rssi) / (10.0 * (indoor ? 4.2 : 2.5))) / 1000.0; }
void handleWelcomeAnimation() { for(int y=SCREEN_HEIGHT; y>=28; y-=2) { display.clearDisplay(); display.setTextSize(2); display.setCursor(24, y); display.print(F("WELCOME")); display.display(); delay(15); } delay(1500); }

void checkButtons() {
  bool L = (digitalRead(BTNS[0]) == LOW), M = (digitalRead(BTNS[1]) == LOW), R = (digitalRead(BTNS[2]) == LOW);
  if (L || M || R) lastActivityTime = millis(); else return;

  if (currentState == STATE_INBOX_TEXT || currentState == STATE_INBOX_VOICE) {
    int p = L ? BTNS[0] : (M ? BTNS[1] : BTNS[2]);
    if (activePressedPin == -1) { activePressedPin = p; buttonPressStartTime = millis(); }
    else if (millis() - buttonPressStartTime >= 2000) {
      if (currentState == STATE_INBOX_TEXT && totalStoredTexts > 0) totalStoredTexts--;
      else { localVoiceNoteExists = isPlayingVoiceNote = activeAudioDataSize = 0; }
      activePressedPin = -1; delay(400);
    }
    return;
  }

  if (R) {
    currentState = (currentState == STATE_SERVER_SELECT) ? STATE_MAIN_HUB : (SystemState)(currentState - 1);
    delay(200);
  } else if (L) {
    if (currentState == STATE_MAIN_HUB) currentHubIdx = (currentHubIdx + 1) % 4;
    if (currentState == STATE_TEXT_CUSTOM) currentLetterIdx = (currentLetterIdx == 0) ? 38 : currentLetterIdx - 1;
    delay(150);
  } else if (M) {
    if (currentState == STATE_MAIN_HUB) currentState = (currentHubIdx == 0) ? STATE_SERVER_SELECT : (currentHubIdx == 1 ? STATE_INBOX_MENU : STATE_HW_CONFIG);
else if (currentState == STATE_ACTION_SELECT) { if(currentActionIdx == 0) recordVoiceNoteBuffer(); else currentState = STATE_TEXT_MENU; }
delay(200);
}
}
void updateUI() {
display.clearDisplay();
if (currentState == STATE_MAIN_HUB) {
drawHeader("OFF-GRID RADIO NETWORK"); drawSelector(currentHubIdx, 4, mainHubOptions);
} else if (currentState == STATE_SERVER_SELECT) {
drawHeader("SELECT SERVER"); drawSelector(currentServerIdx, 3, serverList);
} else if (currentState == STATE_TEXT_CUSTOM) {
drawHeader("TYPE MESSAGE"); display.setCursor(0, 16); display.print(F("> ")); display.print(customMessage);
display.drawFastHLine(0, 36, SCREEN_WIDTH, SSD1306_WHITE); display.setCursor(48, 53); display.print(alphabet[currentLetterIdx]);
} else if (currentState == STATE_HW_CONFIG) {
drawHeader("POWER DIAGNOSTICS"); display.setCursor(0, 14); display.print(F("Power: ")); display.print(currentTxPowerDbm);
}
display.display();
}
void drawHeader(const char* title) { display.setTextSize(1); display.setCursor(0, 0); display.print(title); display.drawFastHLine(0, 11, SCREEN_WIDTH, SSD1306_WHITE); }
void drawSelector(int idx, int total, const char* items[]) {
for(int i=0; i<total; i++) {
int y = 16 + (i * 14); if(i == idx) { display.fillRect(0, y - 2, SCREEN_WIDTH, 11, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); } else display.setTextColor(SSD1306_WHITE);
display.setCursor(4, y); display.print(items[i]);
}
display.setTextColor(SSD1306_WHITE);
}
void drawVoiceUI(unsigned long elapsed) {
drawHeader("RECORDING AUDIO"); display.setCursor(0, 16); display.print((float)elapsed / 1000.0, 1);
display.drawRect(0, 28, SCREEN_WIDTH, 6, SSD1306_WHITE); display.fillRect(0, 28, map(elapsed, 0, 10000, 0, SCREEN_WIDTH), 6, SSD1306_WHITE);
}
