void setup() {
  Serial.begin(115200);
  
  pinMode(BTN_LEFT, INPUT_PULLUP);
  pinMode(BTN_MID, INPUT_PULLUP);
  pinMode(BTN_RIGHT, INPUT_PULLUP);
  pinMode(POT_PIN, INPUT); 
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LORA_RXEN, OUTPUT);
  pinMode(LORA_TXEN, OUTPUT);

  voiceStorageBuffer = (uint8_t*)malloc(AUDIO_BUFFER_SIZE);
  if(voiceStorageBuffer == NULL) {
    Serial.println(F("[ERROR]: Failed to allocate Voice Note RAM cache boundary."));
    while(1);
  }

  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) { 
    Serial.println(F("SSD1306 allocation failed"));
    while(1);
  }
  
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  handleWelcomeAnimation();
  
  initHardwareAudioI2S();

  Serial.print(F("[LORA] Initializing Sub-GHz module... "));
  int state = radio.begin(868.0, 125.0, 9, 7, (uint8_t)0x12, currentTxPowerDbm, 8);
  if (state == RADIOLIB_ERR_NONE) {
    Serial.println(F("Success!"));
    handleTransmitText("[SYS_INIT]: NODE_ONLINE");
    
    radio.startReceive();
    digitalWrite(LORA_RXEN, HIGH); 
    digitalWrite(LORA_TXEN, LOW);
  } else {
    Serial.print(F("Failed, code: ")); Serial.println(state);
    display.clearDisplay(); display.setCursor(0,20); 
    display.print(F("SX1262 Error!\nCheck Bus Wiring.")); 
    display.display();
    while(1);
  }

  lastActivityTime = millis(); 
  currentState = STATE_MAIN_HUB;
}

void loop() {
  checkButtons();
  if (trackingEngineActive) {
    processIncomingRadioTraffic(); 
  }
  readHardwarePotentiometer();
  checkPowerTimeout(); 
  updateUI();
  delay(10); 
}

void initHardwareAudioI2S() {
  Serial.println(F("[I2S INTERFACE] Initializing internal audio buses..."));
  
  i2s_config_t micConfig = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX), 
    .sample_rate = AUDIO_SAMPLE_RATE,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_16BIT,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 4,
    .dma_buf_len = 512,
    .use_apll = false
  };

  i2s_pin_config_t micPins = {
    .bck_io_num = I2S_MIC_SCK,
    .ws_io_num = I2S_MIC_WS,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_MIC_SD
  };

  i2s_driver_install(I2S_NUM_0, &micConfig, 0, NULL);
  i2s_set_pin(I2S_NUM_0, &micPins);
  Serial.println(F("[I2S MIC] INMP441 background logic drivers locked."));
}

void recordVoiceNoteBuffer() {
  Serial.println(F("[I2S RECORD] Opening memory pipeline. Capturing..."));
  size_t bytesRead = 0;
  unsigned long startT = millis();
  activeAudioDataSize = 0;
  
  while(millis() - startT < RECORD_MAX_MS) {
    i2s_read(I2S_NUM_0, (void*)(voiceStorageBuffer + activeAudioDataSize), 1024, &bytesRead, portMAX_DELAY);
    activeAudioDataSize += bytesRead;
    if(activeAudioDataSize >= AUDIO_BUFFER_SIZE - 1024) break;
    
    drawVoiceUI(millis() - startT);
    display.display();
  }
  
  localVoiceNoteExists = true;
  localVoiceSender = "Local Node (Me)";
}

void playActiveVoiceBuffer() {
  if (!localVoiceNoteExists || activeAudioDataSize == 0) return;
  size_t bytesWritten = 0;
  uint32_t playPointer = 0;
  i2s_zero_dma_buffer(I2S_NUM_0);
  
  while(playPointer < activeAudioDataSize && isPlayingVoiceNote) {
    i2s_write(I2S_NUM_0, (void*)(voiceStorageBuffer + playPointer), 1024, &bytesWritten, portMAX_DELAY);
    playPointer += bytesWritten;
    
    display.clearDisplay();
    drawHeader("PLAYING VOICE NOTE");
    display.setCursor(0, 20); display.print(F("Streaming Audio Cache..."));
    for (int x = 0; x < SCREEN_WIDTH; x += 6) {
      int amp = abs(sin((float)(millis() + x) * 0.04) * 8.0);
      display.drawFastVLine(x, SCREEN_HEIGHT - amp, amp, SSD1306_WHITE);
    }
    display.display();
    if(digitalRead(BTN_MID) == LOW) { delay(200); break; }
  }
  isPlayingVoiceNote = false;
  i2s_zero_dma_buffer(I2S_NUM_0);
}

void transmitVoicePacketOverAir() {
  if(activeAudioDataSize == 0) return;
  display.clearDisplay();
  display.setCursor(10, 25); display.print(F("COMPRESSING AUDIO..."));
  display.display();
  
  // FIXED: Allocated a robust 128-byte array to handle media frame headers
  char packetHeader[128];
  snprintf(packetHeader, sizeof(packetHeader), "[V_NOTE:#%s:%u]", espList[currentEspIdx], activeAudioDataSize);
  
  digitalWrite(LORA_RXEN, LOW); digitalWrite(LORA_TXEN, HIGH);
  radio.transmit(packetHeader);
  radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
}

void parseNetworkPacketPayload(String textData, float rssi, float snr) {
  if (textData.indexOf("[SYS_INIT]") >= 0) return;
  
  for(int i = 0; i < totalTrackedDevices; i++) {
Use code with caution.
if(textData.indexOf(trackerRoster[i].id) >= 0 || (i == 0 && textData.indexOf("996") >= 0)) {
trackerRoster[i].liveRssi = rssi;
trackerRoster[i].liveSnr  = snr;
trackerRoster[i].lastSeenMs = millis();
trackerRoster[i].calculatedRadiusKm = calculateRadiusFromRssi(rssi, isIndoorEnvironment);
wakeDisplay();
triggerAlertBeep();
}
}
}
void processIncomingRadioTraffic() {
if (digitalRead(LORA_DIO1) == HIGH) {
String packetStr = "";
packetStr.reserve(64);
int state = radio.readData(packetStr);
if (state == RADIOLIB_ERR_NONE) {
parseNetworkPacketPayload(packetStr, radio.getRSSI(), radio.getSNR());
if (currentState != STATE_MAIL_POPUP && packetStr.indexOf("[SYS_INIT]") < 0) {
prePopupState = currentState;
currentState = STATE_MAIL_POPUP;
popupStartTime = millis();
}
}
radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
}
}
void handleTransmitText(const char* payload) {
display.clearDisplay(); display.setCursor(10, 25); display.print(F("TX FLASH PACKET...")); display.display();
// FIXED: Allocated a large 256-byte stack array to hold outgoing network string wrappers
char structuredPayload[256];
snprintf(structuredPayload, sizeof(structuredPayload), "[SRC:MY_NODE][DST:%s] %s", espList[currentEspIdx], payload);
digitalWrite(LORA_RXEN, LOW); digitalWrite(LORA_TXEN, HIGH);
radio.transmit(structuredPayload);
radio.startReceive(); digitalWrite(LORA_RXEN, HIGH); digitalWrite(LORA_TXEN, LOW);
lastActivityTime = millis();
delay(500);
}
void wakeDisplay() {
if (!isDisplayPoweredOn) { display.ssd1306_command(SSD1306_DISPLAYON); isDisplayPoweredOn = true; }
lastActivityTime = millis();
}
void checkPowerTimeout() {
if (currentState != STATE_MAIL_POPUP && currentState != STATE_WELCOME && isDisplayPoweredOn) {
if (millis() - lastActivityTime >= OLED_TIMEOUT_MS) {
display.ssd1306_command(SSD1306_DISPLAYOFF); isDisplayPoweredOn = false;
}
}
}
void triggerAlertBeep() {
digitalWrite(BUZZER_PIN, HIGH); delay(150); digitalWrite(BUZZER_PIN, LOW); delay(100);
digitalWrite(BUZZER_PIN, HIGH); delay(150); digitalWrite(BUZZER_PIN, LOW);
}
void readHardwarePotentiometer() {
int rawAdc = analogRead(POT_PIN);
int targetPower = map(rawAdc, 0, 4095, 2, 22);
if (targetPower != currentTxPowerDbm) {
currentTxPowerDbm = targetPower;
radio.setOutputPower(currentTxPowerDbm);
if (isIndoorEnvironment) calculatedMaxRangeKm = map(currentTxPowerDbm, 2, 22, 1, 5) / 10.0;
else calculatedMaxRangeKm = map(currentTxPowerDbm, 2, 22, 5, 60) / 10.0;
if(!isDisplayPoweredOn) wakeDisplay();
else lastActivityTime = millis();
}
}
float calculateRadiusFromRssi(float rssi, bool indoor) {
float measuredPower = -42.0; float pathLossExponent = indoor ? 4.2 : 2.5;
if (rssi >= 0) return 0.0;
return pow(10.0, (measuredPower - rssi) / (10.0 * pathLossExponent)) / 1000.0;
}

display.clearDisplay(); display.setTextSize(2); display.setCursor(24, textY); display.print(F("WELCOME")); display.display();

}
delay(INTRO_HOLD_MS);
}
void checkButtons() {
bool leftPressed  = (digitalRead(BTN_LEFT) == LOW);
bool midPressed   = (digitalRead(BTN_MID) == LOW);
bool rightPressed = (digitalRead(BTN_RIGHT) == LOW);
if (!isDisplayPoweredOn && (leftPressed || midPressed || rightPressed)) { wakeDisplay(); delay(300); return; }
if (leftPressed || midPressed || rightPressed) lastActivityTime = millis();
if (currentState == STATE_MAIL_POPUP) return;
if (currentState == STATE_INBOX_TEXT || currentState == STATE_INBOX_VOICE) {
int currentPressedPin = leftPressed ? BTN_LEFT : (midPressed ? BTN_MID : (rightPressed ? BTN_RIGHT : -1));
if (currentPressedPin != -1) {
if (activePressedPin == -1) { activePressedPin = currentPressedPin; buttonPressStartTime = millis(); }
else if (millis() - buttonPressStartTime >= DELETE_HOLD_MS) {
if (currentState == STATE_INBOX_TEXT && totalStoredTexts > 0) { totalStoredTexts--; }
else if (currentState == STATE_INBOX_VOICE && localVoiceNoteExists) { localVoiceNoteExists = false; isPlayingVoiceNote = false; activeAudioDataSize = 0; }
activePressedPin = -1; delay(400); return;
}
} else {
if (activePressedPin != -1 && (millis() - buttonPressStartTime < DELETE_HOLD_MS)) {
if (activePressedPin == BTN_LEFT) {
if (currentState == STATE_INBOX_TEXT) currentTextMsgIdx = (currentTextMsgIdx + 1) % totalStoredTexts;
} else if (activePressedPin == BTN_MID) {
if (currentState == STATE_INBOX_VOICE && localVoiceNoteExists) {
isPlayingVoiceNote = !isPlayingVoiceNote;
if(isPlayingVoiceNote) playActiveVoiceBuffer();
}
} else if (activePressedPin == BTN_RIGHT) { currentState = STATE_INBOX_MENU; isPlayingVoiceNote = false; }
activePressedPin = -1; delay(200);
}
}
return;
}
static bool lastRightState = false;
if (rightPressed && !lastRightState) {
lastRightState = true;
switch(currentState) {
case STATE_SERVER_SELECT:   currentState = STATE_MAIN_HUB; break;
case STATE_ESP_SELECT:      currentState = STATE_SERVER_SELECT; break;
case STATE_ACTION_SELECT:   currentState = STATE_ESP_SELECT; break;
case STATE_VOICE_RECORD:    currentState = STATE_ACTION_SELECT; break;
case STATE_TEXT_MENU:       currentState = STATE_ACTION_SELECT; break;
case STATE_TEXT_CUSTOM:     currentState = STATE_TEXT_MENU; customMessage = ""; customMsgLen = 0; break;
case STATE_INBOX_MENU:      currentState = STATE_MAIN_HUB; break;
case STATE_VOICE_REQUEST:   currentState = STATE_INBOX_VOICE; break;
case STATE_GPS_ENV_PROMPT:  currentState = STATE_MAIN_HUB; break;
case STATE_GPS_MENU:        currentState = STATE_GPS_ENV_PROMPT; break;
case STATE_GPS_ROSTER:      currentState = STATE_GPS_MENU; break;
case STATE_GPS_DETAILS:     currentState = STATE_GPS_ROSTER; break;
case STATE_HW_CONFIG:       currentState = STATE_MAIN_HUB; break;
default: break;
}
delay(200);
} else if (!rightPressed) { lastRightState = false; }
static bool lastLeftState = false;
if (leftPressed && !lastLeftState) {
lastLeftState = true;
switch(currentState) {
case STATE_MAIN_HUB:      currentHubIdx = (currentHubIdx + 1) % totalHubOptions; break;
case STATE_SERVER_SELECT: currentServerIdx = (currentServerIdx + 1) % totalServers; break;
case STATE_ESP_SELECT:    currentEspIdx = (currentEspIdx + 1) % totalEsps; break;
case STATE_ACTION_SELECT: currentActionIdx = (currentActionIdx + 1) % totalActions; break;
case STATE_TEXT_MENU:     currentTextOptIdx = (currentTextOptIdx + 1) % totalTextOptions; break;
case STATE_INBOX_MENU:    currentInboxChoiceIdx = (currentInboxChoiceIdx + 1) % 2; break;
case STATE_VOICE_REQUEST: currentLogIdx = (currentLogIdx + 1) % totalServerLogs; break;
case STATE_GPS_ENV_PROMPT: currentEnvIdx = (currentEnvIdx + 1) % 2; break;
case STATE_GPS_MENU:       currentEnvIdx = (currentEnvIdx + 1) % 2; break;
case STATE_GPS_ROSTER:     currentRosterIdx = (currentRosterIdx + 1) % totalTrackedDevices; break;
case STATE_TEXT_CUSTOM:
if (currentLetterIdx == 0) currentLetterIdx = totalLetters - 1;
else currentLetterIdx--;
break;
default: break;
}
delay(150);
} else if (!leftPressed) { lastLeftState = false; }
if (currentState == STATE_TEXT_CUSTOM) {
if (midPressed) {
if (!midBtnActive) { midBtnActive = true; midBtnPressTime = millis(); }
else if (millis() - midBtnPressTime >= TX_HOLD_MS) {
if (customMessage.length() > 0) { handleTransmitText(customMessage.c_str()); currentState = STATE_ACTION_SELECT; customMessage = ""; customMsgLen = 0; }
midBtnActive = false; delay(400);
}
} else {
if (midBtnActive) {
if (millis() - midBtnPressTime < TX_HOLD_MS) {
if (alphabet[currentLetterIdx] == '<') { if (customMessage.length() > 0) { customMessage.remove(customMessage.length() - 1); } }
else if (customMessage.length() < 30) { customMessage += alphabet[currentLetterIdx]; }
}
midBtnActive = false; delay(200);
}
}
} else {
static bool lastMidState = false;
if (midPressed && !lastMidState) {
lastMidState = true;
switch(currentState) {
case STATE_MAIN_HUB:
if (currentHubIdx == 0) currentState = STATE_SERVER_SELECT;
else if (currentHubIdx == 1) currentState = STATE_INBOX_MENU;
else if (currentHubIdx == 2) { currentState = STATE_GPS_ENV_PROMPT; currentEnvIdx = 0; }
else { currentState = STATE_HW_CONFIG; }
break;
case STATE_SERVER_SELECT: currentState = STATE_ESP_SELECT; break;
case STATE_ESP_SELECT:    currentState = STATE_ACTION_SELECT; break;
case STATE_ACTION_SELECT:
if (currentActionIdx == 0) { currentState = STATE_VOICE_RECORD; recordVoiceNoteBuffer(); currentState = STATE_ACTION_SELECT; transmitVoicePacketOverAir(); }
else currentState = STATE_TEXT_MENU;
break;
case STATE_TEXT_MENU:
if (currentTextOptIdx == 3) { currentState = STATE_TEXT_CUSTOM; currentLetterIdx = 0; }
else { handleTransmitText(textMenuList[currentTextOptIdx]); currentState = STATE_ACTION_SELECT; }
break;
case STATE_INBOX_MENU:
if (currentInboxChoiceIdx == 0) currentState = STATE_INBOX_TEXT;
else currentState = STATE_INBOX_VOICE;
break;
case STATE_VOICE_REQUEST: currentState = STATE_INBOX_MENU; break;
case STATE_GPS_ENV_PROMPT:
isIndoorEnvironment = (currentEnvIdx == 1); currentState = STATE_GPS_MENU; currentEnvIdx = 0;
break;
case STATE_GPS_MENU:
if(currentEnvIdx == 0) { trackingEngineActive = !trackingEngineActive; if(trackingEngineActive) { radio.startReceive(); } }
else { currentState = STATE_GPS_ROSTER; currentRosterIdx = 0; }
break;
case STATE_GPS_ROSTER: currentState = STATE_GPS_DETAILS; break;
default: break;
}
delay(200);
} else if (!midPressed) { lastMidState = false; }
}
}

display.clearDisplay();
switch(currentState) {
case STATE_MAIN_HUB:
drawHeader("OFF-GRID RADIO NETWORK");
for(int i = 0; i < totalHubOptions; i++) {
int yPos = 13 + (i * 12);
if(i == currentHubIdx) { display.fillRect(0, yPos - 1, SCREEN_WIDTH, 11, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); }
else { display.setTextColor(SSD1306_WHITE); }
display.setCursor(4, yPos); display.print(mainHubOptions[i]);
}
display.setTextColor(SSD1306_WHITE); break;
case STATE_SERVER_SELECT:
drawHeader("SELECT SERVER"); drawSelector(currentServerIdx, totalServers, serverList); break;
case STATE_ESP_SELECT:
drawHeader("SELECT NODE"); drawSelector(currentEspIdx, totalEsps, espList); break;
case STATE_ACTION_SELECT:
if (currentEspIdx == 0) drawHeader("TASK: BROADCAST");
else {
// FIXED: Expanded array dimensions to comfortably store standard string formats
char headerBuf[64];
snprintf(headerBuf, sizeof(headerBuf), "TASK: %s", espList[currentEspIdx]);
drawHeader(headerBuf);
}
drawSelector(currentActionIdx, totalActions, actionList); break;
case STATE_VOICE_RECORD: {
drawHeader("RECORDING AUDIO");
break;
}
case STATE_TEXT_MENU:
drawHeader("TEXT PAYLOAD OPT"); drawSelector(currentTextOptIdx, totalTextOptions, textMenuList); break;
case STATE_TEXT_CUSTOM:
drawHeader("TYPE CUSTOM MESSAGE");
display.setTextSize(1); display.setCursor(0, 16); display.print(F("> ")); display.print(customMessage);
if(millis() % 500 < 250) display.print(F("|"));
display.drawFastHLine(0, 36, SCREEN_WIDTH, SSD1306_WHITE);
int prevIdx, nextIdx;
prevIdx = (currentLetterIdx == 0) ? totalLetters - 1 : currentLetterIdx - 1;
nextIdx = (currentLetterIdx == totalLetters - 1) ? 0 : currentLetterIdx + 1;
display.setTextColor(SSD1306_WHITE); display.setCursor(45, 40);
if(alphabet[prevIdx] == '<') display.print(F("<-")); else if(alphabet[prevIdx] == ' ') display.print(F("[SP]")); else display.print(alphabet[prevIdx]);
display.fillRect(40, 50, 26, 13, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); display.setCursor(48, 53);
if(alphabet[currentLetterIdx] == '<') display.print(F("<-")); else if(alphabet[currentLetterIdx] == ' ') display.print(F("[SP]")); else display.print(alphabet[currentLetterIdx]);

break;

break;

display.setCursor(0, 40); display.print(F("[MID] Play Track")); display.setCursor(0, 52); display.print(F("[HOLD ANY BTN TO DEL]"));
}
break;
case STATE_VOICE_REQUEST:
drawHeader("SERVER COPIES AVAILABLE");
for(int i = 0; i < totalServerLogs; i++) {
int yPos = 14 + (i * 15);
if(i == currentLogIdx) { display.fillRect(0, yPos - 1, SCREEN_WIDTH, 14, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); } else { display.setTextColor(SSD1306_WHITE); }

}
display.setTextColor(SSD1306_WHITE); break;
case STATE_GPS_ENV_PROMPT:
drawHeader("ENVIRONMENT SETUP"); display.setCursor(0, 14); display.print(F("Calibrate path loss:"));
for(int i = 0; i < 2; i++) {

else { display.setTextColor(SSD1306_WHITE); }
display.setCursor(10, yPos); display.print(envOptions[i]);
}
display.setTextColor(SSD1306_WHITE); break;
case STATE_GPS_MENU:

if(currentEnvIdx == 0) { display.fillRect(0, 16, SCREEN_WIDTH, 13, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); }

display.setCursor(4, 37); display.print(F("> ARROW TRACK ROSTER")); display.setTextColor(SSD1306_WHITE);

break;
case STATE_GPS_ROSTER:
drawHeader("LIVE PEER CIRCLES");
for (int i = 0; i < totalTrackedDevices; i++) {

if (i == currentRosterIdx) { display.fillRect(0, yPos - 2, SCREEN_WIDTH, 13, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); } else { display.setTextColor(SSD1306_WHITE); }

else { display.setCursor(90, yPos); display.print(trackerRoster[i].calculatedRadiusKm, 2); display.print(F("k")); }
}

case STATE_GPS_DETAILS: {

}
break;
}

drawHeader("POWER & HW DIAGNOSTICS");

display.setCursor(0, 36); display.print(F("Signal Sync : CONNECTED"));
display.setCursor(0, 47); display.print(F("Battery Bank: 94% [STABLE]"));
display.drawFastHLine(0, 56, SCREEN_WIDTH, SSD1306_WHITE); display.setCursor(4, 57);

}
case STATE_MAIL_POPUP: {
unsigned long elapsed = millis() - popupStartTime;
if (elapsed >= POPUP_HOLD_MS) { currentState = prePopupState; lastActivityTime = millis(); }
else {

}
break;
}
}
display.display();
}
void drawHeader(const char* title) {
display.setTextSize(1); display.setTextColor(SSD1306_WHITE); display.setCursor(0, 0); display.print(title);
display.drawFastHLine(0, 11, SCREEN_WIDTH, SSD1306_WHITE);
}
void drawSelector(int currentIdx, int totalItems, const char* items[]) {
display.setTextSize(1);
for(int i = 0; i < totalItems; i++) {
int yPos = 16 + (i * 14);
if(i == currentIdx) { display.fillRect(0, yPos - 2, SCREEN_WIDTH, 11, SSD1306_WHITE); display.setTextColor(SSD1306_BLACK); }
else { display.setTextColor(SSD1306_WHITE); }
display.setCursor(4, yPos); display.print(items[i]);
}
display.setTextColor(SSD1306_WHITE);
}
void drawVoiceUI(unsigned long elapsed) {
display.clearDisplay();
drawHeader("RECORDING AUDIO");
display.setTextSize(1); display.setCursor(0, 16); display.print(F("Time: ")); display.print((float)elapsed / 1000.0, 1); display.print(F("s / 10s"));
int progressWidth = map(elapsed, 0, RECORD_MAX_MS, 0, SCREEN_WIDTH);
display.drawRect(0, 28, SCREEN_WIDTH, 6, SSD1306_WHITE); display.fillRect(0, 28, progressWidth, 6, SSD1306_WHITE);
display.setCursor(0, 40); display.print(F("Audio Wave:"));
for (int x = 0; x < SCREEN_WIDTH; x += 4) {
int amplitude = abs(sin((float)(millis() + x) * 0.2) * 12.0);
display.drawFastVLine(x, SCREEN_HEIGHT - amplitude, amplitude, SSD1306_WHITE);
}
